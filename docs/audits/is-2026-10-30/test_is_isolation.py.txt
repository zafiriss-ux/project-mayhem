def _decision(day: str, athlete: str, session: str = "EASY") -> dict:
    return {
        "athlete_profile_id": athlete, "date": day, "session_planned": session,
        "session_post_governance": session, "locks_applied": [], "governance_clean": True,
        "phase": "BASE", "vlg_color": "GREEN", "apl_priority_vector": {},
        "lock_thresholds_active": {}, "acwr": 1.1, "tsb_value": -5.0,
        "readiness_score": 70.0, "glycogen_debt": 1.0,

# ── reads and writes, two athletes ───────────────────────────────────────────

def test_daily_metrics_two_athletes_see_only_their_own(db):
    db.save_daily_metric(date=TODAY, hrv=50.0, rhr=50, athlete_id="owner")
    db.save_daily_metric(date=TODAY, hrv=90.0, rhr=40, athlete_id=A1)
    assert [v for _, v in db.get_hrv_history(days=7, athlete_id=A1)] == [90.0]
    assert [v for _, v in db.get_hrv_history(days=7, athlete_id="owner")] == [50.0]
    assert db.get_hrv_history(days=7, athlete_id=A2) == []
    assert db.get_daily_metric(TODAY, athlete_id=A1)["resting_hr"] == 40


def test_training_and_load_two_athletes_see_only_their_own(db):
    db.save_session(date=TODAY, session_type="EASY", distance_km=5, duration_sec=1800, rpe=5,
                    athlete_id="owner", garmin_activity_id="g-owner")
    db.save_session(date=TODAY, session_type="LONG", distance_km=20, duration_sec=7200, rpe=6,
                    athlete_id=A1, garmin_activity_id="g-a1")
    assert len(db.get_recent_activities(athlete_id=A1)) == 1
    assert sum(v for _, v in db.get_training_load_history(7, athlete_id=A2)) == 0
    assert db.get_ctl_atl_tsb(athlete_id=A2) == (0.0, 0.0, 0.0) or db.get_ctl_atl_tsb(athlete_id=A2)[0] == 0


def test_defect_same_garmin_activity_id_from_another_athlete_is_silently_dropped(db):
    # D-079: garmin_activity_id is unique across the DB. A second athlete's row with the same id
    # is ignored with no error (INSERT OR IGNORE) and the first athlete's row stays untouched.
    db.save_session(date=TODAY, session_type="EASY", distance_km=5, duration_sec=1800, rpe=5,
                    athlete_id=A1, garmin_activity_id="g-same")
    db.save_session(date=TODAY, session_type="EASY", distance_km=9, duration_sec=3600, rpe=5,
                    athlete_id=A2, garmin_activity_id="g-same")
    assert len(db.get_recent_activities(athlete_id=A1)) == 1
    assert db.get_recent_activities(athlete_id=A2) == []  # no error raised, A2's session lost


def test_governance_two_athletes_same_day_isolated(db):
    db.save_governance_decision(_decision(TODAY, "owner", "TEMPO"))
    db.save_governance_decision(_decision(TODAY, A1, "REST"))
    mine = db.get_governance_decisions(athlete_id=A1)
    assert [d["session_post_governance"] for d in mine] == ["REST"]
    assert all(d["athlete_profile_id"] == A1 for d in mine)
    assert db.annotate_governance_outcome(TODAY, "note-a1", athlete_id=A1) is True
    assert db.update_governance_outcome(TODAY, rpe=9, athlete_id=A1) is True
    owner = db.get_governance_decision_on("owner", TODAY)
    assert not owner.get("outcome_notes") and owner.get("outcome_rpe") is None
    assert db.get_governance_decisions(athlete_id=A2) == []

@pytest.mark.parametrize("unknown", ["a-ffffffff", "OWNER", " owner", "nobody"])
def test_defect_unknown_athlete_id_is_accepted_on_writes(db, unknown):
    # Expected per §5.2: an unknown id ends in denial. Observed: the rows are written.
    db.save_daily_metric(date=TODAY, hrv=70.0, athlete_id=unknown)
    db.save_session(date=TODAY, session_type="EASY", distance_km=3, duration_sec=900, rpe=3,
                    athlete_id=unknown)
    db.save_governance_decision(_decision(TODAY, unknown))
    db.log_override(date=TODAY, prescribed="EASY", actual="TEMPO", reason="x", athlete_id=unknown)
    with db._lock:
        known = {r[0] for r in db.conn.execute("SELECT athlete_id FROM athletes")}

    assert unknown not in known

    assert len(db.get_governance_decisions(athlete_id=unknown)) == 1
    # Not the owner's data either way: the owner sees nothing of it.
    assert db.get_daily_metric(TODAY, athlete_id="owner") is None

def _cookie(c):
    from titan.api.auth import ACCESS_COOKIE_NAME
    return c.cookies.get(ACCESS_COOKIE_NAME)


def test_tampered_and_foreign_tokens_are_refused(api):
    import jwt as pyjwt
    from cryptography.hazmat.primitives import serialization
    from cryptography.hazmat.primitives.asymmetric import rsa

    from titan.api.auth import ACCESS_COOKIE_NAME, JWT_AUDIENCE, JWT_ISSUER
    app, _ = api
    c, _ = _client(app, "ath1")
    token = _cookie(c)
    head, body, sig = token.split(".")
    claims = json.loads(base64.urlsafe_b64decode(body + "=" * (-len(body) % 4)))
    # 1) body changed to another athlete, old signature
    forged = dict(claims, athlete_id=A2, sub="u-own")
    fbody = base64.urlsafe_b64encode(json.dumps(forged).encode()).rstrip(b"=").decode()
    fresh = __import__("fastapi.testclient", fromlist=["TestClient"]).TestClient(app, base_url="https://testserver")
    fresh.cookies.set(ACCESS_COOKIE_NAME, f"{head}.{fbody}.{sig}")
    assert fresh.get(f"/athletes/{A2}/dashboard").status_code == 401
    # 2) signed with another key
    other = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    pem = other.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8,
                              serialization.NoEncryption())
    fresh.cookies.set(ACCESS_COOKIE_NAME, pyjwt.encode(forged, pem, algorithm="RS256"))
    assert fresh.get("/athletes/owner/dashboard").status_code == 401
    # 3) alg=none
    none_tok = pyjwt.encode(forged, None, algorithm="none") if hasattr(pyjwt, "encode") else ""
    fresh.cookies.set(ACCESS_COOKIE_NAME, none_tok)
    assert fresh.get("/athletes/owner/dashboard").status_code == 401
    # 4) a valid token whose claims say COACH for an ATHLETE user: the DB role wins
    from titan.api.auth import _jwt_private_key
    coach_claims = dict(claims, role="COACH", athlete_id=A2,
                        exp=int((datetime.now(UTC) + timedelta(minutes=5)).timestamp()),
                        iss=JWT_ISSUER, aud=JWT_AUDIENCE)
    fresh.cookies.set(ACCESS_COOKIE_NAME, pyjwt.encode(coach_claims, _jwt_private_key(), algorithm="RS256"))
    assert fresh.get(f"/athletes/{A2}/dashboard").status_code == 403
    # 5) a valid token for a user that does not exist
    ghost = dict(coach_claims, sub="ghost")
    fresh.cookies.set(ACCESS_COOKIE_NAME, pyjwt.encode(ghost, _jwt_private_key(), algorithm="RS256"))
    assert fresh.get(f"/athletes/{A1}/dashboard").status_code == 401
