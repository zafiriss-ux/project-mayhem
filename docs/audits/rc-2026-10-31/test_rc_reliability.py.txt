def _db(path: Path):
    from titan.core.database import DatabaseManager

    return DatabaseManager(db_name="titan_core.db", db_dir=path)


def _child(code: str, *args: str, wait: bool = True, timeout: float = 120):
    script = textwrap.dedent(code)
    env = {**os.environ, "PYTHONPATH": str(REPO), "TELEGRAM_BOT_TOKEN": ""}
    proc = subprocess.Popen([PY, "-c", script, *args], cwd=str(REPO), env=env,
                            stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    if not wait:
        return proc
    out, err = proc.communicate(timeout=timeout)
    return proc.returncode, out, err

        athlete_id="owner", execution_date=day, routine_type="morning", started_at=now,

        assert len(runs) == 1, outs
        assert all(o[0] in ("run", "wait") for o in outs), outs
    print("CLAIM ROUNDS", results)

def test_defect_a_live_second_runner_of_the_same_kind_is_treated_as_restarted(tmp_path) -> None:
    """Two runners of one kind (two consoles, or a worker beside a stray worker without the
    flock) are told apart only by process token; a fresh mark of the other is treated as
    'that process restarted', so a second attempt starts while the first still runs.
    Fencing then stops the first from sending (M0-01); the window M0-01-R1 remains."""
    db = _db(tmp_path)
    try:
        kw = dict(athlete_id="owner", execution_date=DAY, routine_type="morning", max_attempts=3,
                  retry_gap=timedelta(minutes=15), stale_after=timedelta(minutes=30))
        assert db.begin_routine_attempt(started_at=T0, runner="console:AAA", **kw) == ("run", 1)
        # 16 minutes later, attempt 1 is still running (mark 16 min old, budget 30 min).
        second = db.begin_routine_attempt(started_at=T0 + timedelta(minutes=16), runner="console:BBB", **kw)
        other_kind = db.begin_routine_attempt(started_at=T0 + timedelta(minutes=16), runner="worker:CCC", **kw)
        print("SAME-KIND", second, "OTHER-KIND", other_kind)
        assert second == ("run", 2)  # the live attempt 1 is superseded
        assert other_kind == ("wait", 2)
    finally:
        db.close()

def _hold_write_lock(path: Path, seconds: float, ready: threading.Event) -> None:
    conn = sqlite3.connect(str(path / "titan_core.db"), timeout=0.1)
    conn.isolation_level = None
    conn.execute("BEGIN IMMEDIATE")
    ready.set()
    time.sleep(seconds)
    conn.execute("ROLLBACK")
    conn.close()


def test_holds_claim_fails_closed_when_the_database_is_locked(tmp_path, bot) -> None:
    import titan.worker as worker

    db = _db(tmp_path)
    try:
        db.conn.execute("PRAGMA busy_timeout=300")
        ready = threading.Event()
        t = threading.Thread(target=_hold_write_lock, args=(tmp_path, 3.0, ready))
        t.start()
        ready.wait(5)
        started = time.monotonic()
        result = worker._claim_routine(db, "owner", T0, spaced=True, last_start=False)
        took = time.monotonic() - started
        t.join()
        rows = db.conn.execute("SELECT COUNT(*) FROM worker_routine_executions").fetchone()[0]
        print("LOCKED CLAIM ->", result, f"{took:.2f}s", "rows", rows)
        assert result is None and rows == 0
        assert worker._claim_routine(db, "owner", T0, spaced=True, last_start=False) == 1
    finally:
        db.close()

def test_holds_a_migration_that_raises_midway_leaves_nothing_and_the_restart_completes(tmp_path, monkeypatch) -> None:
    from titan.core.database import SCHEMA_VERSION, DatabaseManager

    original = DatabaseManager._apply_v34_migrations

    def _boom(self, cursor):
        original(self, cursor)
        raise RuntimeError("synthetic crash inside v34")

    monkeypatch.setattr(DatabaseManager, "_apply_v34_migrations", _boom)
    with pytest.raises(RuntimeError, match="synthetic crash"):
        _db(tmp_path)
    conn = sqlite3.connect(tmp_path / "titan_core.db")
    left = (conn.execute("PRAGMA user_version").fetchone()[0],
            conn.execute("SELECT COUNT(*) FROM sqlite_master WHERE type='table'").fetchone()[0])
    conn.close()
    monkeypatch.setattr(DatabaseManager, "_apply_v34_migrations", original)
    db = _db(tmp_path)
    version = db.conn.execute("PRAGMA user_version").fetchone()[0]
    db.close()
    print("AFTER RAISE (user_version, tables):", left, "restart ->", version)
    assert left == (0, 0) and version == SCHEMA_VERSION


def test_holds_a_migration_killed_midway_leaves_the_old_state(tmp_path) -> None:
    code = """
import sys, time
from pathlib import Path
from titan.core.database import DatabaseManager
original = DatabaseManager._apply_v34_migrations
def slow(self, cursor):

DatabaseManager(db_name="titan_core.db", db_dir=Path(sys.argv[1]))

def _fill(db, rows: int) -> None:

        for i in range(rows):

            )


def test_defect_a_backup_killed_midway_blocks_that_days_backup_and_is_kept(tmp_path, caplog) -> None:
    """The worker is killed during the daily hot backup (docker stop past its grace, OOM).
    The half-written titan_<today>.db stays. The next call sees 'today's backup already
    done', returns None without checking it, logs nothing, and retention keeps it."""
    db = _db(tmp_path)
    _fill(db, 40000)
    db.close()
    backups = tmp_path / "backups"
    code = """
import sys, time
from pathlib import Path
from titan.core.database import DatabaseManager
db = DatabaseManager(db_name="titan_core.db", db_dir=Path(sys.argv[1]))
real = db.conn
class Slow:
    def __getattr__(self, name):
        return getattr(real, name)
    def backup(self, target, **kw):
        def progress(status, remaining, total):
            print("PROGRESS", remaining, total, flush=True)
            time.sleep(0.2)
        real.backup(target, pages=4, progress=progress)
db.conn = Slow()
db.backup_database(backup_dir=Path(sys.argv[2]))
"""
    proc = _child(code, str(tmp_path), str(backups), wait=False)
    for _ in range(3):
        print(proc.stdout.readline().strip())
    os.kill(proc.pid, signal.SIGKILL)
    proc.wait(10)
    today = datetime.now(UTC).strftime("%Y-%m-%d")
    partial = backups / f"titan_{today}.db"
    assert partial.exists()
    size_before = partial.stat().st_size

    db = _db(tmp_path)
    caplog.set_level(logging.INFO)
    try:
        result = db.backup_database(backup_dir=backups, keep=7, keep_weekly=4)
    finally:
        db.close()
    from titan.core.database import SCHEMA_VERSION, _verify_backup_artifact

    problem = _verify_backup_artifact(partial, expected_user_version=SCHEMA_VERSION)
    critical = [r for r in caplog.records if r.levelno >= logging.CRITICAL]
    sys.path.insert(0, str(REPO / "scripts"))
    import restore_drill

    drill = restore_drill.run_drill(backups)
    print("SECOND CALL ->", result, "| partial size", size_before, "| verify:", problem,
          "| CRITICAL logs:", len(critical), "| corrupt files:", list(backups.glob("*.corrupt")),
          "| drill:", drill)
    assert result is None  # "already done"
    assert partial.exists() and not list(backups.glob("*.corrupt")) and not critical
    assert problem is not None  # the file kept as today's backup is not a valid database
    assert drill["ok"] is False  # the drill catches it, if someone runs it


def test_defect_runbook_restore_step_copies_the_db_without_its_wal(tmp_path) -> None:
    """RUNBOOK_DR step 3: `cp titan_core.db titan_core.db.before-restore` then rm the -wal.
    After a SIGKILL (docker stop past grace, OOM) committed rows live only in the -wal."""
    code = """
import os, signal, sys
from pathlib import Path
from titan.core.database import DatabaseManager
db = DatabaseManager(db_name="titan_core.db", db_dir=Path(sys.argv[1]))
db.conn.execute("INSERT INTO daily_metrics (date, athlete_id, hrv_rmssd, resting_hr) VALUES ('2026-10-29', 'owner', 51, 49)")
db.conn.commit()
os.kill(os.getpid(), signal.SIGKILL)
"""
    rc, _out, err = _child(code, str(tmp_path))
    wal = tmp_path / "titan_core.db-wal"
    wal_size = wal.stat().st_size if wal.exists() else 0
    shutil.copy2(tmp_path / "titan_core.db", tmp_path / "titan_core.db.before-restore")
    conn = sqlite3.connect(f"file:{tmp_path / 'titan_core.db.before-restore'}?mode=ro", uri=True)
    try:
        rows = conn.execute("SELECT COUNT(*) FROM daily_metrics WHERE athlete_id='owner'").fetchone()[0]
    except sqlite3.DatabaseError as exc:
        rows = f"error: {exc}"
    conn.close()
    live = _db(tmp_path)
    live_rows = live.conn.execute("SELECT COUNT(*) FROM daily_metrics WHERE athlete_id='owner'").fetchone()[0]
    live.close()
    print("KILLED rc", rc, "| wal bytes", wal_size, "| rows in before-restore copy:", rows, "| rows in live db+wal:", live_rows)
    assert rc == -signal.SIGKILL and wal_size > 0
    assert rows != live_rows

class _TimeoutAfterDeliveryTelegram:
    """Telegram took the message, but the reply timed out: send_message returns False."""

def test_defect_m0_01_r1_fence_check_and_send_are_separate_steps(tmp_path) -> None:
    """The fence is read, then Telegram is called. A takeover between the two is not seen."""
    import titan.worker as worker
    from titan.orchestration.daily_coach import DailyCoach

    db = _db(tmp_path)
    try:
        kw = dict(athlete_id="owner", execution_date=DAY, routine_type="morning", max_attempts=3,
                  runner="worker:A", stale_after=timedelta(minutes=30))
        assert db.begin_routine_attempt(started_at=T0, **kw) == ("run", 1)
        sent: list = []

        class _Tg:
            def send_daily_coaching(self, payload):
                # Between the fence check and this call, attempt 2 takes the day over.
                db.begin_routine_attempt(started_at=T0 + timedelta(minutes=31), **{**kw, "runner": "console:B"})
                sent.append(payload)
                return True

                                attempt_is_current=lambda: worker._attempt_is_current(db, "owner", DAY, 1))
        ok = DailyCoach._make_send_telegram_handler(coach)(None, SimpleNamespace(coaching_payload={"session_type": "EASY"}))
        current = worker._attempt_is_current(db, "owner", DAY, 1)
        print("SENT by superseded attempt:", ok, len(sent), "attempt 1 still current:", current)
        assert ok is True and len(sent) == 1 and current is False
    finally:
        db.close()

def test_defect_activity_day_from_start_time_local_is_the_local_day(tmp_path) -> None:
    """Athens is UTC+2 after 25 Oct 2026. A run at 01:30 local on 31 Oct is 23:30 UTC on 30 Oct.
    Without startTimeGMT the day is the local one; with it, the UTC one."""
    from titan.integrations.garmin_sync import _activity_day

    local_only = _activity_day({"startTimeLocal": "2026-10-31 01:30:00"})
    with_gmt = _activity_day({"startTimeLocal": "2026-10-31 01:30:00", "startTimeGMT": "2026-10-30 23:30:00"})
    print("DAY local-only:", local_only, "with GMT:", with_gmt)
    assert local_only == "2026-10-31" and with_gmt == "2026-10-30"


class _FrozenDatetime(datetime):
    frozen: datetime = datetime(2026, 10, 30, 22, 30, tzinfo=UTC)

    @classmethod
    def now(cls, tz=None):
        return cls.frozen if tz is None else cls.frozen.astimezone(tz)


def test_defect_local_day_dates_are_rejected_as_future_near_utc_midnight(monkeypatch) -> None:
    """22:30 UTC on 30 Oct = 00:30 in Athens on 31 Oct. A date-only '2026-10-31' (local day,
    the startTimeLocal fallback; HRV calendarDate) is read as 00:00 UTC, 1.5 h ahead: rejected."""
    import titan.science.data_sanity as ds

    monkeypatch.setattr(ds, "datetime", _FrozenDatetime)
    checker = ds.DataSanityChecker()
    act = checker.check_activity({"date": "2026-10-31", "distance_km": 5.0, "duration_sec": 1800,
                                  "avg_hr": 140, "max_hr": 160, "activity_id": "x1"})
    hrv = checker.check_hrv({"date": "2026-10-31", "rmssd": 50.0})
    rules = sorted({v.rule for v in act + hrv})
    print("REJECTED at 22:30 UTC:", rules)
    assert "timestamp_in_future" in rules


def test_defect_naive_full_timestamp_is_read_as_utc(monkeypatch) -> None:
    """A naive local time (UTC+3 summer) of 09:30 is 06:30 UTC; at 07:00 UTC it is in the past.
    Read as UTC it is 2.5 h ahead: rejected. Latent: the real sync passes date-only values."""
    import titan.science.data_sanity as ds

    _FrozenDatetime.frozen = datetime(2026, 7, 1, 7, 0, tzinfo=UTC)
    monkeypatch.setattr(ds, "datetime", _FrozenDatetime)
    try:
        hrv = ds.DataSanityChecker().check_hrv({"date": "2026-07-01T09:30:00", "rmssd": 50.0})
    finally:
        _FrozenDatetime.frozen = datetime(2026, 10, 30, 22, 30, tzinfo=UTC)
    rules = [v.rule for v in hrv]
    print("NAIVE LOCAL 09:30 at 07:00 UTC ->", rules)
    assert "timestamp_in_future" in rules


def test_holds_the_scheduler_is_utc_across_the_dst_change(monkeypatch) -> None:
    """25 Oct 2026, Europe/Athens goes from UTC+3 to UTC+2. The wait to 07:00 is computed in UTC."""
    import titan.worker as worker

    class _Clock(datetime):
        @classmethod
        def now(cls, tz=None):
            return datetime(2026, 10, 25, 0, 30, tzinfo=UTC)

    monkeypatch.setattr(worker._dt, "datetime", _Clock)
    monkeypatch.setenv("TZ", "Europe/Athens")
    time.tzset()
    try:
        wait = worker._seconds_until_next_run(7, 0)
    finally:
        monkeypatch.delenv("TZ")
        time.tzset()
    print("WAIT on DST night (s):", wait)
    assert wait == 6.5 * 3600

def test_defect_a_gap_longer_than_the_14_day_sync_is_never_backfilled(tmp_path) -> None:
    """Worker down 20 days. The next morning syncs 14 days. Backfill runs only when no
    session is older than 42 days, so the activities of days 15-20 back are never fetched."""
    from titan.orchestration.garmin_persistence import backfill_training_history

    db = _db(tmp_path)
    calls: list = []
    try:
        today = datetime(2026, 10, 31).date()
        for back in (60, 50, 45, 30, 25):  # history before the outage
            db.conn.execute(
                "INSERT INTO training_log (date, session_type, distance_km, duration_sec, training_load, "
                "athlete_id, garmin_activity_id) VALUES (?, 'EASY', 8, 2700, 60, 'owner', ?)",
                ((today - timedelta(days=back)).isoformat(), f"g{back}"),
            )
        db.conn.commit()
        garmin = SimpleNamespace(athlete_max_hr=190, sync_activities=lambda *a, **k: calls.append(a) or [])
        fetched = backfill_training_history(db, garmin, today.isoformat(), athlete_id="owner",
                                            emit=lambda *a, **k: None)
        print("BACKFILL fetched", fetched, "garmin calls", calls)
        assert fetched == 0 and calls == []
    finally:
        db.close()


def test_defect_local_restore_scenario_b_keeps_the_old_wal_beside_the_restored_file(tmp_path) -> None:
    """docs/runbooks/database-recovery.md Scenario B: `cp backups/titan_core.db.YYYYMMDD titan_core.db`,
    with no removal of titan_core.db-wal / -shm. After a crash the old WAL is replayed onto the
    restored file: rows that are not in the backup appear, or the file is damaged."""
    live = tmp_path / "live"
    live.mkdir()
    db = _db(live)
    db.conn.execute("INSERT INTO daily_metrics (date, athlete_id, hrv_rmssd) VALUES ('2026-10-01', 'owner', 40)")
    db.conn.commit()
    backup = tmp_path / "titan_core.db.20261001"
    db.conn.execute("PRAGMA wal_checkpoint(TRUNCATE)")
    with sqlite3.connect(str(backup)) as dst:
        db.conn.backup(dst)
    db.close()
    # The installation runs on, then crashes with committed rows only in the WAL.
    code = """
import os, signal, sys
from pathlib import Path
from titan.core.database import DatabaseManager
db = DatabaseManager(db_name="titan_core.db", db_dir=Path(sys.argv[1]))
db.conn.execute("PRAGMA wal_autocheckpoint=0")
for d in range(2, 20):
    db.conn.execute("INSERT INTO daily_metrics (date, athlete_id, hrv_rmssd) VALUES (?, 'owner', 99)", (f"2026-10-{d:02d}",))
db.conn.commit()
os.kill(os.getpid(), signal.SIGKILL)
"""
    rc, _o, _e = _child(code, str(live))
    wal = live / "titan_core.db-wal"
    wal_size = wal.stat().st_size if wal.exists() else 0
    # Scenario B, step by step: copy the backup over the database, nothing else.
    shutil.copy2(backup, live / "titan_core.db")
    conn = sqlite3.connect(str(live / "titan_core.db"))
    try:
        integrity = conn.execute("PRAGMA integrity_check").fetchone()[0]
        rows = conn.execute("SELECT COUNT(*) FROM daily_metrics").fetchone()[0]
    except sqlite3.DatabaseError as exc:
        integrity, rows = f"error: {exc}", None
    conn.close()
    bconn = sqlite3.connect(f"file:{backup}?mode=ro", uri=True)
    backup_rows = bconn.execute("SELECT COUNT(*) FROM daily_metrics").fetchone()[0]
    bconn.close()
    print("CRASH rc", rc, "| old wal bytes", wal_size, "| restored file: integrity", integrity,
          "rows", rows, "| rows in the backup itself", backup_rows)
    assert rc == -signal.SIGKILL and wal_size > 0
    assert rows != backup_rows or integrity != "ok"


def test_measure_the_effect_of_a_lost_6_day_block_on_ctl_atl_tsb() -> None:
    """Synthetic athlete: 60 load every day for 210 days. Worker down 20 days; the activities of
    days 15-20 back are never fetched (previous test). Compare the kernel with and without them."""
    from datetime import date

    from titan.science.tsb_engine import compute_ctl_atl_tsb

    today = date(2026, 10, 31)
    full = [((today - timedelta(days=b)).isoformat(), 60.0) for b in range(1, 211)]
    gap = [(d, v) for d, v in full if not 15 <= (today - date.fromisoformat(d)).days <= 20]
    with_all = compute_ctl_atl_tsb(full, reference_date=today)
    with_gap = compute_ctl_atl_tsb(gap, reference_date=today)
    acwr = lambda c: round(c[1] / c[0], 3) if c[0] else None  # noqa: E731
    print("ALL (ctl, atl, tsb)", with_all, "acwr", acwr(with_all), "| GAP", with_gap, "acwr", acwr(with_gap))
    assert with_gap[0] < with_all[0]  # CTL lower
    assert with_gap[2] <= with_all[2]  # TSB not higher: the error points the conservative way here
