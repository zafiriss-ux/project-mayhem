"""SD audit, Deliverable 1: gate matrix H1–H5 through the real DailyCoach morning.

Each test asserts the EXPECTED behaviour written in the report table before the run
(ADR-0009, ADR-0051, ADR-0069, CONTRACT 7.9 §13 core rule). A failing test is a MISMATCH.
Every test prints the actual outcome (run with -s). Synthetic data only.
"""
from __future__ import annotations

import math
import sqlite3
from datetime import UTC, datetime, timedelta

import pytest

from harness import Scenario, findings, run_morning, summary

ABOVE_EASY = {"TEMPO", "INTERVALS", "PROGRESSIVE", "HAMMER", "LONG_RUN", "RACE"}


def _show(tag, res):
    print(f"\n[{tag}] {summary(res)}")


def _closed(res) -> bool:
    return res.payload.get("session_type") in {"EASY", "REST"}


# ── H1 ─────────────────────────────────────────────────────────────────────

def test_h1_missing_is_unknown_and_told(tmp_path, monkeypatch):
    """H1 no source: wording UNKNOWN (ADR-0069 §3); the athlete is told it was not measured."""
    res = run_morning(tmp_path, monkeypatch, Scenario(fatigue={0: 2}))
    _show("H1-missing", res)
    assert findings(res)["H1"] == "UNKNOWN"
    reason = (res.governed.get("decision_rationale") or {}).get("primary_reason", "")
    assert "H1" in reason, "the message does not say H1 was not measured"


def test_h1_open_injury_rest(tmp_path, monkeypatch):

    _show("H1-open", res)

    assert "HARD_STOP_SAFETY_GATES" in res.payload["governance_locks"]
    assert findings(res)["H1"] == "TRIGGERED"

def _heavy_loads():
    loads = {i: 40 for i in range(8, 120)}
    loads.update({i: 140 for i in range(2, 8)})
    return loads


def _yesterday_run(load=320):
    yday = (datetime.now(UTC).date() - timedelta(days=1)).isoformat()
    return [{"date": yday, "activity_type": "running", "activity_id": "syn-1", "distance_km": 30.0,
             "duration_sec": 9000, "avg_hr": 160, "max_hr": 180, "estimated_training_load": load,
             "name": "synthetic"}]


def test_h4_reference_heavy_yesterday_triggers(tmp_path, monkeypatch):
    """Reference: yesterday's heavy run synced → TSB ≤ −50 → H4."""
    res = run_morning(tmp_path, monkeypatch, Scenario(fatigue={0: 2}, loads=_heavy_loads(),
                                                      activities=_yesterday_run()))
    _show("H4-ref", res)
    assert findings(res)["H4"] == "TRIGGERED"


def test_h4_load_refresh_failure_not_clear(tmp_path, monkeypatch):
    """M0-22: the reload after the sync fails → TSB without yesterday's run; must not be CLEAR."""
    def _patch(coach, db, mp):
        def _boom(*_a, **_k):
            raise sqlite3.OperationalError("synthetic")
        mp.setattr(db, "get_training_load_history", _boom)
    res = run_morning(tmp_path, monkeypatch, Scenario(fatigue={0: 2}, loads=_heavy_loads(),
                                                      activities=_yesterday_run(), patch=_patch))
    _show("H4-refreshfail", res)
    assert findings(res)["H4"] != "CLEAR", "silent load-refresh failure gave H4 CLEAR"


def test_h4_activity_sync_lost_not_clear(tmp_path, monkeypatch):
    """Activity endpoint failed (sync returns no activities) while HRV/sleep synced.

    The morning cannot tell 'no run' from 'run not synced'; expected: load not measured → not CLEAR.
    """
    res = run_morning(tmp_path, monkeypatch, Scenario(fatigue={0: 2}, loads=_heavy_loads(), activities=[]))
    _show("H4-activitieslost", res)
    assert findings(res)["H4"] != "CLEAR", "unsynced load counted as rest; H4 CLEAR"


def test_h4_contradictory_low_tsb_high_readiness(tmp_path, monkeypatch):
    res = run_morning(tmp_path, monkeypatch, Scenario(fatigue={0: 2}, loads=_heavy_loads(),
