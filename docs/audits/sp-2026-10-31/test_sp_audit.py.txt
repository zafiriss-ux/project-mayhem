def _rsa_pair() -> tuple[str, str]:
    key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    priv = key.private_bytes(
        serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8, serialization.NoEncryption()
    ).decode()
    pub = key.public_key().public_bytes(
        serialization.Encoding.PEM, serialization.PublicFormat.SubjectPublicKeyInfo
    ).decode()
    return priv, pub

def test_holds_coach_routes_scoped_and_duplicate_query_consistent(api):
    c = api["client"]()
    _login(c, "coach_a")
    ok = c.post("/coach/overrides?athlete_id=a-00000001", json={})
    bad = c.post("/coach/overrides?athlete_id=a-00000002", json={})
    # duplicate parameter: the dependency and the handler must see the same value
    dup = c.post("/coach/overrides?athlete_id=a-00000001&athlete_id=a-00000002", json={})
    dup2 = c.post("/coach/notes?athlete_id=a-00000002&athlete_id=a-00000001", json={})
    empty = c.post("/coach/notes?athlete_id=", json={})
    none = c.post("/coach/notes", json={})
    print("COACH ROUTES", ok.status_code, bad.status_code, dup.status_code, dup.text[:80],
          dup2.status_code, dup2.text[:80], empty.status_code, none.status_code)
    assert ok.status_code == 200 and bad.status_code == 403
    assert dup.status_code == 403  # last value a-00000002 is checked and refused
    assert dup2.status_code == 200 and dup2.json()["athlete_id"] == "a-00000001"  # same value both sides
    assert empty.status_code == 403
    assert none.status_code in (403, 422)

def test_defect_registered_secret_leaks_when_json_escaped():
    """Structured events are json.dumps'd before logging (and _telegram_event redacts after
    json.dumps). A registered secret with a quote, backslash or non-ASCII character is escaped
    by json.dumps, so str.replace no longer finds it."""
    from titan.utils.secrets import RedactingFilter, redact_secrets, register_secret

    pw = 'Κωδ"ικός\\canary-77'  # synthetic: Greek letters, a quote and a backslash
    register_secret(pw)
    assert pw not in redact_secrets(f"login failed with {pw}")  # plain form: redacted

    line = json.dumps({"flow": "garmin", "action": "failure", "error_message": f"login failed with {pw}"})
    out = redact_secrets(line)
    rec = logging.LogRecord("titan.x", logging.ERROR, "", 0, line, None, None)
    RedactingFilter().filter(rec)
    escaped = json.dumps(pw)[1:-1]
    print("JSON LINE AFTER REDACTION:", out)
    assert escaped in out and escaped in rec.getMessage()
    # repr() form (exception args in a traceback or %r) leaks too
    rep = redact_secrets(repr(ValueError(f"bad password {pw}")))
    print("REPR AFTER REDACTION:", rep)
    assert "canary-77" in rep


def test_defect_api_process_installs_no_redaction(env, tmp_path):
    """scripts/run_api.py composes the API with log_config=None and never calls
    install_log_redaction: no handler and no RedactingFilter, so a token in a log line
    reaches stderr unchanged (M0-17)."""
    code = textwrap.dedent(
        f"""
        import logging, runpy, sys
        sys.argv = ["run_api.py"]
        ns = runpy.run_path({str(REPO / "scripts" / "run_api.py")!r}, run_name="not_main")
        from titan.utils.secrets import RedactingFilter
        root = logging.getLogger()
        on_root = any(isinstance(f, RedactingFilter) for h in root.handlers for f in h.filters)
        on_url = any(isinstance(f, RedactingFilter) for f in logging.getLogger("urllib3.connectionpool").filters)
        print("ROOT_HANDLERS", len(root.handlers), "FILTER_ON_ROOT", on_root, "FILTER_ON_URLLIB3", on_url)
        logging.getLogger("titan.api").warning("readyz db check failed: OperationalError: {CANARY_TOKEN}")
        """
    )
    r = subprocess.run(
        [sys.executable, "-c", code], cwd=tmp_path, env={**os.environ, "PYTHONPATH": str(REPO)},
        capture_output=True, text=True, timeout=120,
    )
    print("API PROC stdout:", r.stdout.strip(), "| stderr tail:", r.stderr.strip().splitlines()[-1:])
    assert r.returncode == 0, r.stderr
    assert "FILTER_ON_ROOT False" in r.stdout and "FILTER_ON_URLLIB3 False" in r.stdout
    assert CANARY_TOKEN in r.stderr

def test_defect_privacy_test_ignores_event_kwargs():
    """tests/test_log_privacy.py looks only at extra={...} keys. An event kwarg such as
    chat_id=... passes it: the scan below finds one in titan/ that the test does not see."""
    found = []
    keys = {"chat_id", "email", "password", "name", "display_name", "birth_date", "telegram_chat_id"}
    for p in sorted((REPO / "titan").rglob("*.py")):
        for n in ast.walk(ast.parse(p.read_text(encoding="utf-8"))):
            if isinstance(n, ast.Call):
                fn = getattr(n.func, "id", None) or getattr(n.func, "attr", None) or ""
                if fn.endswith("_event") or fn == "structured_event_payload":
                    found += [f"{p.relative_to(REPO)}:{n.lineno}:{kw.arg}" for kw in n.keywords if kw.arg in keys]
    print("EVENT KWARGS WITH PERSONAL KEYS:", found)
    assert found

def test_holds_consent_gate_fail_closed(env):
    from datetime import UTC, datetime, timedelta

    from titan.core.database import DatabaseManager

    db = DatabaseManager("titan_core.db", db_dir=env["data"])
    try:
        aid = "a-0000abcd"
        db.conn.execute(
            "INSERT INTO athletes (athlete_id, display_name, created_at, is_cold_start, cold_start_sessions_count)"
            " VALUES (?, 'S', datetime('now'), 1, 0)", (aid,))
        db.conn.commit()
        now = datetime(2026, 10, 31, 9, tzinfo=UTC)
        no = db.has_current_consent(aid, now=now)
        db.grant_consent(aid, "v1", now=now)
        yes = db.has_current_consent(aid, now=now)
        db.revoke_consent(aid)
        after_stop = db.has_current_consent(aid, now=datetime.now(UTC))
        db.grant_consent(aid, "v1", now=datetime.now(UTC))
        same_day = db.has_current_consent(aid, now=datetime.now(UTC))
        next_day = db.has_current_consent(aid, now=datetime.now(UTC) + timedelta(days=1))
        with pytest.raises(ValueError):
            db.grant_consent(aid, "v0")
        print("CONSENT none/granted/after-stop/re-same-day/re-next-day:", no, yes, after_stop, same_day, next_day)
        assert (no, yes, after_stop, same_day, next_day) == (False, True, False, False, True)
    finally:
        db.close()
